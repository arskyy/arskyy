import Me from "about-me"

const amar = new Me()

const introduce = async () => {
    const profile = {
        name: "Amar",
        role: "Student",
        hobby: "Playing Music"
    }

    return amar.sendMessage(
        profile,
        amar.message
    )
}

await introduce()
